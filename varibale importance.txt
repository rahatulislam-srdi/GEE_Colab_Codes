var table = ee.FeatureCollection("projects/ee-rahat5151/assets/Habiganj_Sadar"),
    table2 = ee.FeatureCollection("projects/ee-rahat5151/assets/Habiganj_Sadar_River"),
    table3 = ee.FeatureCollection("projects/ee-rahat5151/assets/Habiganj_Sadar_Roads");

var roi = table;
var rivers = table2;

var studyArea = typeof roi !== 'undefined' ? roi : rivers.geometry().bounds();
Map.centerObject(studyArea, 11); 


var rivers = ee.FeatureCollection('projects/ee-rahat5151/assets/Habiganj_Sadar_River').filterBounds(roi);
var riverRaster = ee.Image().byte().paint(rivers, 1).clip(roi);
var riverDist = riverRaster.fastDistanceTransform(30).sqrt().multiply(30).rename('RIVER_DIST').clip(roi);


var roads = ee.FeatureCollection('projects/ee-rahat5151/assets/Habiganj_Sadar_Roads').filterBounds(roi);
var roadsRaster = ee.Image().byte().paint(roads, 1).clip(roi);
var roadsDist = roadsRaster.fastDistanceTransform(30).sqrt().multiply(30).rename('ROAD_DIST').clip(roi); // Corrected Name


var dem = ee.Image('USGS/SRTMGL1_003').clip(roi);
var slope = ee.Terrain.slope(dem).rename('SLOPE');


function scaleLandsat(img){
  return img.multiply(2.75e-05).add(-0.2);
}


var ls1993 = ee.ImageCollection("LANDSAT/LT05/C02/T1_L2")
  .filterDate('1993-08-01', '1993-12-31')
  .filterBounds(roi)
  .filter(ee.Filter.lt('CLOUD_COVER', 45))
  .median();
ls1993 = scaleLandsat(ls1993);

var ndvi93 = ls1993.normalizedDifference(['SR_B4', 'SR_B3']).rename('NDVI');
var ndbi93 = ls1993.normalizedDifference(['SR_B5', 'SR_B4']).rename('NDBI');
var wiw93  = ls1993.select('SR_B4').lte(0.1558).and(ls1993.select('SR_B5').lte(0.0871)).rename('WIW').clip(roi);


var ls2023 = ee.ImageCollection("LANDSAT/LC08/C02/T1_L2")
  .filterDate('2023-08-01', '2023-12-31')
  .filterBounds(roi)
  .filter(ee.Filter.lt('CLOUD_COVER', 45))
  .median();
ls2023 = scaleLandsat(ls2023);

var ndvi23 = ls2023.normalizedDifference(['SR_B5', 'SR_B4']).rename('NDVI');
var ndbi23 = ls2023.normalizedDifference(['SR_B6', 'SR_B5']).rename('NDBI');
var wiw23  = ls2023.select('SR_B5').lte(0.1735).and(ls2023.select('SR_B6').lte(0.1035)).rename('WIW').clip(roi);


var ndviChange = ndvi23.subtract(ndvi93).rename('ndvi_ch').clip(roi);
var ndbiChange = ndbi23.subtract(ndbi93).rename('ndbi_ch').clip(roi);
var wiwChange  = wiw23.unmask(0).subtract(wiw93.unmask(0)).rename('wiw_ch').clip(roi);

var predictors = ndviChange
  .addBands(ndbiChange)
  .addBands(dem.rename('dem'))
  .addBands(slope.rename('slope'))
  .addBands(roadsDist)
  .addBands(riverDist);


var wetlandLostBinary = wiw23.unmask(0).eq(0).and(wiw93.unmask(0).eq(1)).rename('lost_class');

var trainingPoints = wetlandLostBinary.stratifiedSample({
  numPoints: 1000,
  classBand: 'lost_class',
  region: roi,
  scale: 30,
  geometries: true
});

var trainingData = predictors.sampleRegions({
  collection: trainingPoints,
  properties: ['lost_class'],
  scale: 30
});

var withRandom = trainingData.randomColumn('random');
var trainSet = withRandom.filter(ee.Filter.lt('random', 0.8));
var testSet  = withRandom.filter(ee.Filter.gte('random', 0.8));


var rfClassifier = ee.Classifier.smileRandomForest({
    numberOfTrees: 200
})
.setOutputMode('PROBABILITY')
.train({
    features: trainSet,
    classProperty: 'lost_class',
    inputProperties: predictors.bandNames()
});



var explain = rfClassifier.explain();
var rawImportance = ee.Dictionary(explain.get('importance'));
var totalSum = rawImportance.values().reduce(ee.Reducer.sum());

var relativeImportance = rawImportance.map(function(key, val) {
  return ee.Number(val).divide(totalSum).multiply(100);
});

print('Gini Variable Importance (%):', relativeImportance);

var fc = ee.FeatureCollection(
  relativeImportance.keys().zip(relativeImportance.values()).map(function(kv){
    kv = ee.List(kv);
    return ee.Feature(null, {
      'variable': kv.get(0),
      'importance_pct': kv.get(1)
    });
  })
);

var chart = ui.Chart.feature.byFeature({
  features: fc,
  xProperty: 'variable',
  yProperties: ['importance_pct']
})
.setChartType('ColumnChart')
.setOptions({
  title: 'Variable Importance (%) - Gini Method',
  hAxis: {title: 'Predictors'},
  vAxis: {title: 'Importance (%)', minValue: 0},
  legend: {position: 'none'},
  colors: ['#d95f02']
});

print(chart);
