import ee

ee.Authenticate()
ee.Initialize(
    project = 'ee-rahat5151',
    opt_url = 'https://earthengine-highvolume.googleapis.com'
)



import geemap

Cumilla_SD= (
    ee.FeatureCollection('projects/ee-rahat5151/assets/Cumilla_SD')
)


map = geemap.Map()
map


map.addLayer(Cumilla_SD)


roi=Cumilla_SD.geometry()
roi


forest = (
    ee.Image("UMD/hansen/global_forest_change_2025_v1_13")
    .select('loss', 'lossyear')
)
loss_area = forest.select('loss').eq(1).And(forest.select('lossyear').eq(15)
)



modis = (
    ee.ImageCollection("MODIS/061/MOD13Q1")
    .filterBounds(roi)
    .filterDate('2020','2025')
    .select ('NDVI')

)
modis



from xee import helpers
from shapely.geometry import shape
import xarray as xr


grid_params = helpers.fit_geometry(
    geometry = shape(roi.getInfo()),
    grid_crs = 'EPSG:3857',
    grid_scale = (250, -250)
)


ds = xr.open_dataset(
    modis,
    engine = 'ee',
    **grid_params
)
ds

ds= ds.sortby('time')*0.0001


ds_mean =ds.mean(dim = ['x','y']).to_dataframe()


ds_mean.plot()


from statsmodels.tsa.seasonal import STL


stl_model = STL(ds_mean,period = 24)
stl_result = stl_model.fit()


ndvi_trend = stl_result.trend
ndvi_trend.plot()
