var roi = ee.FeatureCollection("projects/ee-rahat5151/assets/Habiganj_Sadar");

Map.centerObject(roi, 10);
Map.addLayer(roi, {}, "ROI");

function scaleLandsat(img){
  return img.multiply(2.75e-05).add(-0.2);
}

var roiGeom = ee.Geometry(roi.geometry());
var grid = roiGeom.coveringGrid('EPSG:4326', 10000); 
var gridWithRandom = grid.randomColumn('rand_grid');

var trainGrid = gridWithRandom.filter(ee.Filter.lt('rand_grid', 0.7));
var testGrid  = gridWithRandom.filter(ee.Filter.gte('rand_grid', 0.7));

var ls1993 = ee.ImageCollection("LANDSAT/LT05/C02/T1_L2")
  .filterDate('1993-08-01', '1993-12-31')
  .filterBounds(roi)
  .filter(ee.Filter.lt('CLOUD_COVER', 30
  .median();

ls1993 = scaleLandsat(ls1993);

var mndwi93 = ls1993.normalizedDifference(['SR_B2','SR_B5']).rename('MNDWI');


var wiw93   = ls1993.select('SR_B4').lte(0.1558)
                  .and(ls1993.select('SR_B7').lte(0.0871))
                  .rename('WIW');


var ls2023 = ee.ImageCollection("LANDSAT/LC08/C02/T1_L2")
  .filterDate('2023-08-01', '2023-12-31')
  .filterBounds(roi)
  .filter(ee.Filter.lt('CLOUD_COVER', 30))
  .median();

ls2023 = scaleLandsat(ls2023);


var mndwi23 = ls2023.normalizedDifference(['SR_B3','SR_B6']).rename('MNDWI');
var wiw23   = ls2023.select('SR_B5').lte(0.1735)
                  .and(ls2023.select('SR_B7').lte(0.1035))
                  .rename('WIW');

var mndwiClass93 = mndwi93.gt(0).rename('class');
var wiwClass93   = wiw93.rename('class');
var mndwiClass23 = mndwi23.gt(0).rename('class');
var wiwClass23   = wiw23.rename('class');

function printFormattedMetrics(cm, testSamples, modelName, year, indexName) {
  var oa = cm.accuracy();
  var kappa = cm.kappa();
  

  var pa = cm.producersAccuracy();
  var ua = cm.consumersAccuracy();
  var cmArray = cm.array();
  var tn = ee.Number(cmArray.get([0, 0]));
  var fp = ee.Number(cmArray.get([0, 1]));
  var fn = ee.Number(cmArray.get([1, 0]));
  var tp = ee.Number(cmArray.get([1, 1]));
  var precision = tp.divide(tp.add(fp));
  var recall    = tp.divide(tp.add(fn)); 
  var f1Score   = ee.Number(2).multiply(precision.multiply(recall)).divide(precision.add(recall));


  var n = testSamples.size();
  var se = oa.multiply(ee.Number(1).subtract(oa)).divide(n).sqrt();
  var ci_margin = ee.Number(1.96).multiply(se);
  var ci_lower = oa.subtract(ci_margin);
  var ci_upper = oa.add(ci_margin);


  var summary = ee.Dictionary({
    '1_Year': year,
    '2_Index': indexName,
    '3_Model': modelName,
    '4_Overall_Accuracy_%': oa.multiply(100),
    '5_95%_CI_Lower_%': ci_lower.multiply(100),
    '6_95%_CI_Upper_%': ci_upper.multiply(100),
    '7_Kappa': kappa,
    '8_PA_Water_%': ee.Number(pa.get([1,0])).multiply(100),
    '9_UA_Water_%': ee.Number(ua.get([0,1])).multiply(100),
    'A1_Precision_Water': precision,
    'A2_Recall_Water': recall,
    'A3_F1_Score_Water': f1Score
  });

  print('=== ' + year + ' | ' + indexName + ' | ' + modelName + ' ===');
  print('Confusion Matrix (0: Non-Water, 1: Water):', cm);
  print('Metrics Summary Table Row:', summary);
}

function runML(image, classImg, bands, year, indexName) {

  var stack = image.select(bands).addBands(classImg);

  var trainSamples = stack.sample({
    region: trainGrid,
    scale: 30,
    numPixels: 4200,
    seed: 1
  });

  var testSamples = stack.sample({
    region: testGrid,
    scale: 30,
    numPixels: 1800,
    seed: 1
  });

  var cart = ee.Classifier.smileCart().train(trainSamples, 'class', bands);
  var cartMap = image.select(bands).classify(cart);
  var cartCM  = testSamples.classify(cart).errorMatrix('class','classification');
  printFormattedMetrics(cartCM, testSamples, 'CART', year, indexName);

  var rf = ee.Classifier.smileRandomForest(50).train(trainSamples, 'class', bands);
  var rfMap = image.select(bands).classify(rf);
  var rfCM  = testSamples.classify(rf).errorMatrix('class','classification');
  printFormattedMetrics(rfCM, testSamples, 'Random Forest', year, indexName);

  var svm = ee.Classifier.libsvm().train(trainSamples, 'class', bands);
  var svmMap = image.select(bands).classify(svm);
  var svmCM  = testSamples.classify(svm).errorMatrix('class','classification');
  printFormattedMetrics(svmCM, testSamples, 'SVM', year, indexName);

  var vis = {min:0, max:1, palette:['brown','blue']};
  Map.addLayer(cartMap.clip(roi), vis, year + ' ' + indexName + ' CART');
  Map.addLayer(rfMap.clip(roi),   vis, year + ' ' + indexName + ' RF');
  Map.addLayer(svmMap.clip(roi),  vis, year + ' ' + indexName + ' SVM');
}


runML(ls1993, mndwiClass93, ['SR_B2','SR_B4','SR_B5','SR_B7'], '1993', 'MNDWI');
runML(ls1993, wiwClass93,   ['SR_B2','SR_B4','SR_B5','SR_B7'], '1993', 'WIW');
runML(ls2023, mndwiClass23, ['SR_B3','SR_B5','SR_B6','SR_B7'], '2023', 'MNDWI');
runML(ls2023, wiwClass23,   ['SR_B3','SR_B5','SR_B6','SR_B7'], '2023', 'WIW');
