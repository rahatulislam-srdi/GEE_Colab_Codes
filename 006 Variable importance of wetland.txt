var wetlandLostBinary = wiw2023.eq(1).rename('lost_class');

var trainingPoints = wetlandLostBinary.stratifiedSample({
  numPoints: 1000,          
  classBand: 'lost_class',  
  region: roi,
  scale: 30, 
  geometries: true
});

var trainingData = predictors.sampleRegions({
  collection: trainingPoints, 
  properties: ['lost_class'], 
  scale: 30
});

var withRandom = trainingData.randomColumn('random');
var trainSet = withRandom.filter(ee.Filter.lt('random', 0.8));
var testSet = withRandom.filter(ee.Filter.gte('random', 0.8));

var rfClassifier = ee.Classifier.smileRandomForest({
    numberOfTrees: 200 
})
.setOutputMode('PROBABILITY') 
.train({
    features: trainSet,        
    classProperty: 'lost_class', 
    inputProperties: predictors.bandNames() 
});

var risk = predictors.classify(rfClassifier).rename('risk');

var riskVis = {

  ]
};

var explain = rfClassifier.explain();
var importance = ee.Dictionary(explain.get('importance'));


print('Variable Importance (Raw Weights):', importance);

var keys = importance.keys();
var values = importance.values();


var fc = ee.FeatureCollection(
  keys.zip(values).map(function(kv){
    kv = ee.List(kv);
    return ee.Feature(null, {
      variable: kv.get(0),
      importance: kv.get(1)
    });
  })
);

var totalImportance = values.reduce(ee.Reducer.sum());

var normalizedValues = values.map(function(val) {
  return ee.Number(val).divide(totalImportance).multiply(100);
});

var fcNormalized = ee.FeatureCollection(
  keys.zip(normalizedValues).map(function(kv){
    kv = ee.List(kv);
    return ee.Feature(null, {
      variable: kv.get(0),
      importance_pct: kv.get(1)
    });
  })
);

var chartPct = ui.Chart.feature.byFeature({
  features: fcNormalized,
  xProperty: 'variable',
  yProperties: ['importance_pct']
})
.setChartType('ColumnChart')
.setOptions({
  title: 'Variable Importance (Percentage - Total 100%)',
  hAxis: {title: 'Predictors'},
  vAxis: {title: 'Importance (%)', minValue: 0, maxValue: 100},
  legend: {position: 'none'},
  colors: ['#2b8cbe']
});

print(chartPct);

var importanceDictPct = ee.Dictionary.fromLists(keys, normalizedValues);
print('Variable Importance (%):', importanceDictPct);
