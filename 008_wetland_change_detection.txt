var roi = ee.FeatureCollection("projects/ee-rahat5151/assets/Habiganj_Sadar");

Map.centerObject(roi, 10);
Map.addLayer(roi, {}, "Study Area (ROI)", false);

function scaleLandsat(img){
  var bands = img.select('SR_B.*').multiply(0.0000275).add(-0.2);
  return img.addBands(bands, null, true);
}
var ls1993 = ee.ImageCollection("LANDSAT/LT05/C02/T1_L2")
  .filterDate('1993-08-01', '1993-12-31')
  .filterBounds(roi)
  .filter(ee.Filter.lt('CLOUD_COVER', 30))
  .median();
ls1993 = scaleLandsat(ls1993);


var ls2023 = ee.ImageCollection("LANDSAT/LC08/C02/T1_L2")
  .filterDate('2023-08-01', '2023-12-31')
  .filterBounds(roi)
  .filter(ee.Filter.lt('CLOUD_COVER', 30))
  .median();
ls2023 = scaleLandsat(ls2023);

var wiw93 = ls1993.select('SR_B4').lte(0.1558)
                .and(ls1993.select('SR_B7').lte(0.0871))
                .rename('WIW_1993').clip(roi).gt(0);

var wiw23 = ls2023.select('SR_B5').lte(0.1735)
                .and(ls2023.select('SR_B7').lte(0.1035))
                .rename('WIW_2023').clip(roi).gt(0);
var wiwChange = wiw23.subtract(wiw93).rename('WIW_Change');

var wetlandLoss = wiwChange.eq(-1);
var wetlandGain = wiwChange.eq(1);

Map.addLayer(wiw93.updateMask(wiw93), {palette: ['blue']}, 'WIW Water 1993', false);
Map.addLayer(wiw23.updateMask(wiw23), {palette: ['cyan']}, 'WIW Water 2023', false);
Map.addLayer(wetlandLoss.updateMask(wetlandLoss), {palette: ['red']}, 'Gross Wetland Loss');
Map.addLayer(wetlandGain.updateMask(wetlandGain), {palette: ['green']}, 'Gross Wetland Gain');


var areaImage = ee.Image.pixelArea().divide(1e6); 
function calculateBinaryArea(binaryImg, year, indexName) {
  var wetland = binaryImg.eq(1);
  var upland = binaryImg.eq(0);
  
  var wetlandArea = areaImage.updateMask(wetland).reduceRegion({
    reducer: ee.Reducer.sum(), geometry: roi, scale: 30, maxPixels: 1e13
  });
  
  var uplandArea = areaImage.updateMask(upland).reduceRegion({
    reducer: ee.Reducer.sum(), geometry: roi, scale: 30, maxPixels: 1e13
  });

  print(year + ' ' + indexName + ' Area Breakdown (Sq. Km):', {
    'Wetland (Water/Marshy)': wetlandArea.values().get(0),
    'Upland / Non-Wetland': uplandArea.values().get(0)
  });
}

function calculateChangeArea() {
  var lossArea = areaImage.updateMask(wetlandLoss).reduceRegion({
    reducer: ee.Reducer.sum(), geometry: roi, scale: 30, maxPixels: 1e13
  });
  
  var gainArea = areaImage.updateMask(wetlandGain).reduceRegion({
    reducer: ee.Reducer.sum(), geometry: roi, scale: 30, maxPixels: 1e13
  });

  print('------------------ WETLAND DYNAMICS (1993 - 2023) ------------------');
  print('Gross Wetland Loss (Sq. Km):', lossArea.values().get(0));
  print('Gross Wetland Gain (Sq. Km):', gainArea.values().get(0));
}

print('------------------ AREA ANALYSIS ------------------');

calculateBinaryArea(wiw93, '1993', 'WIW');
calculateBinaryArea(wiw23, '2023', 'WIW');
calculateChangeArea();
